字符驱动
字符设备驱动程序可以分为三个部分：
自动配置和初始化子程序：负责检测所要驱动的硬件设备是否存在和是否能正常工作
服务于I/O请求的子程序：驱动程序的上半部分，在这部分程序执行的时候，系统依然认为是和进行调用的进程属于同一个进程，只是由用户态变成了核心态
中断服务子程序：驱动程序的下半部分

驱动程序上半部分需要实现的系统调用接口通过在Linux内核源代码的include/linux下的fs.h中定义了如下的结构体来表示:

字符设备驱动程序的架构:
a. 头文件: Linux内核源代码中的头文件
b. 声明许可协议: MODULE_LICENSE("GPL");
c. 注册和注销:
   初始化函数init：static int __init myinit(void)
   释放函数exit: static void __exit myexit(void)
   在驱动中通过module_init()和module_exit()关联具体的初始化和释放函数
d. 定义一个struct file_operations结构体类型的变量，关联IO操作和驱动程序中对该操作的具体定义(针对该设备对应的操作需要完成的任务)
e. 实现IO接口

内核空间和用户空间数据的交互:
unsigned long copy_from_user(void *to, const void *from, unsigned long n);  //完成用户空间到内核空间
unsigned long copy_to_user (void * to, void * from, unsigned long len);   //内核空间到用户空间
raw_copy_from_user, raw_copy_to_user

驱动程序实例:
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/fs.h> //struct file_operations
#include <linux/init.h>
#include <linux/types.h>
#include <asm/uaccess.h>

#include MAJOR_NUM 238

//定义一个模拟设备信息的数据
static int dev_value = 99;  //内核空间中的数据

MODULE_LICENSE("GPL");

//定义具体的IO实现：read和write
//read(fd, buf, len)
/*
unsigned long copy_from_user(void *to, const void *from, unsigned long n);  //完成用户空间到内核空间
unsigned long copy_to_user (void * to, void * from, unsigned long len);   //内核空间到用户空间
*/
ssize_t test_char_read (struct file *fp, char __user *buf, size_t len, loff_t *t)
{
	//将内核空间中dev_value的值写入到用户空间中的buf
	int length = raw_copy_to_user(buf, &dev_value, 4);

	return length;
}

ssize_t test_char_write (struct file *fp, const char __user *buf, size_t len, loff_t *t)
{
	//将用户空间中的数据写入到内核空间中驱动程序里面的变量
	int length = raw_copy_from_user(&dev_value, buf, 4);

	return length;
}

//定义一个设备的IO接口 -- 设备和设备文件对应操作的关联
static struct file_operations myfops = 
{
	owner:THIS_MODULE,
	read:test_char_read, //系统调用read的时候，具体的write的操作是由test_char_read完成的
	write:test_char_write, //系统调用write的时候，具体的write的操作是由test_char_write完成的
};

//初始化函数
static int __init myinit(void)
{
	//完成设备的初始化: 硬件初始化和数据初始化
	//向操作系统注册
	/*
	int register_chrdev(unsignedintmajor,constchar*name,structfile_operations*fops);
	是Linux内核中用于注册字符设备驱动程序的核心函数，通过指定主设备号（）、设备名称（）及文件操作结构体（）实现设备驱动注册。当为0时内核动态分配主设备号，非零值则进行静态注册，注册成功后可通过查看设备信息。
	*/
	if(register_chrdev(MAJOR_NUM, "Test_Char_Dev_0819", &myfops))
	{
		//注册失败
		printk("<4>Register Failed\n");
		return -1;
	}
	else
	{
		//注册成功
		printk("<4>Register Success\n");
		return 0;
	}

	return 0;
}

//释放函数
static void __exit myexit(void)
{
	printk("<4>Test_Char_Dev_0819 exit\n");
	/*
	unregister_chrdev_region是Linux内核用于释放字符设备号的系统函数，其原型为void unregister_chrdev_region(dev_t from, unsigned count)，用于注销从from开始的count个设备号
	*/
	unregister_chrdev(MAJOR_NUM, "Test_Char_Dev_0819");
}

module_init(myinit);
module_exit(myexit);
