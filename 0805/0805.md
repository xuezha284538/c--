###  模糊查询
使用 like 关键字 ,配合_ % 来查询
    % 任意字符
    _ 一个字符
```sql
select name from staff where name ="xuezha";

select name from staff where name like 'a%'; --a 开头
select name from staff where name like '%a'; --a 结尾
select name from staff where name like '%a%'; -- 包含a
select name from staff where name like '_a%'; -- 第二个是a
select name from staff where name like '__a%'; -- 第三个是a






```
