# 数据库
## DDL    数据定义语言 操作表格用table
- `CREATE`      创建
- `DROP`        删除
- `ALTER`       修改
- `TRUNCATE`    截断
- `GRANT`       授权
- `...`


```sql
create database udb;
use udb;
show tables;

create table t1
(
    id int auto_increment primary key,
    c2 char(4),
    c3 decimal(7,2)
);

alter table t1 add c4 varchar(16);  /*添加一列*/
alter table t1 CHANGE c2 title varchar(4);/*修改*/
alter table `t1` CHANGE `c4` `order`varchar(4);/*修改*/
alter table t1 DROP COLUMN c3; /*删除*/

insert into  t1(`title`) value("5s4");
insert into  t1(`title`) value("5w4");
insert into  t1(`title`) value("5e4");
/*先drop 在create 保留了表结构 删除表里数据 释放空间 重置自动增长的序列*/
-- 
TRUNCATE TABLE t1; /*清空表中数据*/
--删除表
DROP TABLE t1;
--删除数据
delete from t1 where xxx;
delete from t1 where id =3;
--


--grant 新建用户 % 任意ip
CREATE USER 'ABC'@'%'IDENTIFIED BY '123456';
--授权最高权限
GRANT ALL PRIVILEGES ON *.* TO 'xuezha'@'%' WITH GRANT OPTION;
--*.*：所有数据库的所有数据表；
--ALL PRIVILEGES：全部权限；
--WITH GRANT OPTION：该账号还可以给别的用户授权。

--连接客户端
mysql -u xuezha -p
--授权 数据库权限
grant all ON udb .* TO 'xuezha'@'%';
grant all ON mall .* TO 'xuezha'@'%';

```
