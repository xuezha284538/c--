### Singleton 单例
- 类只能创建一个对象
- 在任意位置获得这个类的实例都是同一个

实现：
- 构造方法私有化，封装，屏蔽了访问方法
- 提供一个类接口，获得唯一实例（对象）

```c
// singleton.cxx
#include <iostream>

using namespace std;

class Singleton
{
private:
    Singleton() {
        cout << "构造函数" << endl;
    }      // 构造函数私有化

public:
    ~Singleton() {}

    // 饿汉式：调用之前，提取初始化了资源
    static Singleton instance;

    // 开放一个静态接口
    static Singleton *getInstance()
    {
        // 懒汉式：lazy 延迟加载初始化 第一次调用才创建对象，分配空间
        // static Singleton instance;      // 类变量，一份
        return &instance;               // 返回的同一份
    }
};

// 类变量的初始化
Singleton Singleton::instance;

int main(int argc, char const *argv[])
{

    // Singleton *s2 = Singleton::getInstance();
    // Singleton *s3 = Singleton::getInstance();
```

## 工厂模式

- 简单/静态工厂
- 抽象工厂

```c
// factory.cxx
#include <iostream>
#include <string>

using namespace std;

class Doc
{
protected:
    string name;

public:
    Doc(string name) : name(name) {}
    ~Doc() {}
    void open()
    {
        cout << "OPEN: " << name << endl;
    }
    virtual void edit() = 0;
};

class Word : public Doc
{
public:
    Word(string name) : Doc(name) {}
    ~Word() {}
    void edit() override
    {
        cout << "EDIT Word: " << name << endl;
    }
};

class Ppt : public Doc
{
public:
    Ppt(string name) : Doc(name) {}
    ~Ppt() {}
    void edit() override
    {
        cout << "EIDT ppt: " << name << endl;
    }
};

class Excel : public Doc
{
public:
    Excel(string name) : Doc(name) {}
    ~Excel() {}
    void edit()
    {
        cout << "EDIT excel: " << name << endl;
    }
};

enum DocType
{
    WORD,
    PPT,
    EXCEL,
    PDF
};

class WPS
{
public:
    WPS() {}
    ~WPS() {}

    static Doc *create(DocType type, string name)
    {
        Doc *doc;
        switch (type)
        {
        case WORD:
            doc = new Word(name);
            break;
        case PPT:
            doc = new Ppt(name);
            break;
        case EXCEL:
            doc = new Excel(name);
            break;
        default:
            break;
        }
        return doc;
    }
};

int main(int argc, char const *argv[])
{
    Doc *d;
    // 静态方法
    d = WPS::create(WORD, "简历");
    d = WPS::create(PPT, "答辩");
    // d->open();
    d->edit();
    // d->open();
    d->edit();

    return 0;
}

```
